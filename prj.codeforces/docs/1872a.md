#include <iostream>

int main() {
    int t;
    std::cin >> t;

    int a;
    int b;
    int c;

    for (int i = 0; i < t; i++) {
        std::cin >> a;
        std::cin >> b;
        std::cin >> c;

        int count = 0;
        int diff;

        if (a > b) {
            diff = a - b;
        }
        else {
            diff = b - a;
        }

        while (diff > 0) {
            diff = diff - 2 * c;
            count += 1;
        }

        std::cout << count;
    }

    return 0;
}
